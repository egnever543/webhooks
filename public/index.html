Enviar um evento 
curl -X POST "$APP/api/ingest/acme/site-vendas" \
  -H "X-Webhook-Secret: <segredo-do-projeto>" \
  -H "X-Event-Type: order.paid" \
  -H "Content-Type: application/json" \
  -d '{"order_id":123,"total":49.9}'
